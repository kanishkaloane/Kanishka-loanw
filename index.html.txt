        <!-- ඔයාගේ Form එක (Form ID එක 'loanForm' ලෙස තබා ඇත) -->
        <form id="loanForm">
            <label>සම්පූර්ණ නම:</label>
            <input type="text" id="name" placeholder="ඔබේ නම ඇතුලත් කරන්න">

            <label>දුරකථන අංකය:</label>
            <input type="tel" id="phone" placeholder="උදා: 0769233462">

            <label>NIC (හැඳුනුම්පත් අංකය):</label>
            <input type="text" id="nic" placeholder="උදා: 991234567V හෝ 199912345678">

            <label>ණය මුදල:</label>
            <select id="amountSelect">
                <option value="">මුදල තෝරන්න</option>
                <option value="50000">රු. 50,000</option>
                <option value="100000">රු. 100,000</option>
                <option value="200000">රු. 200,000</option>
            </select>

            <label>බැංකුව:</label>
            <select id="bank">
                <option value="">බැංකුව තෝරන්න</option>
                <option value="BOC">ලංකා බැංකුව (BOC)</option>
                <option value="People's Bank">මහජන බැංකුව</option>
                <option value="Commercial Bank">කොමර්ෂල් බැංකුව</option>
            </select>

            <button type="submit">අයදුම්පත යොමු කරන්න</button>
        </form>

    <!-- ඔයාගේ වැඩේ කරන ප්‍රධාන JavaScript (Script) කොටස -->
    <script>
        document.getElementById('loanForm').addEventListener('submit', function(e) {
            e.preventDefault();

            // දත්ත ලබා ගැනීම
            let name = document.getElementById('name').value.trim();
            let phone = document.getElementById('phone').value.trim();
            let nic = document.getElementById('nic').value.trim().toUpperCase();
            let amount = document.getElementById('amountSelect').value;
            let bank = document.getElementById('bank').value;

            // 1. හිස් දත්ත පරීක්ෂාව
            if (!name || !phone || !nic || !amount || !bank) {
                Swal.fire({
                    icon: 'warning',
                    title: 'අසම්පූර්ණයි!',
                    text: 'කරුණාකර සියලුම විස්තර ඇතුලත් කරන්න.',
                    confirmButtonColor: '#3085d6'
                });
                return;
            }

            // 2. දුරකථන අංක පරීක්ෂාව
            let phonePattern = /^(0)?7[0-9]{8}$/;
            if (!phonePattern.test(phone)) {
                Swal.fire({
                    icon: 'error',
                    title: 'වැරදි දුරකථන අංකයක්!',
                    text: 'කරුණාකර නිවැරදි දුරකථන අංකයක් ඇතුලත් කරන්න. (උදා: 0769233462)',
                    confirmButtonColor: '#d33'
                });
                return;
            }

            // 3. NIC අංක පරීක්ෂාව
            let nicPattern = /^([0-9]{9}[VX]|[0-9]{12})$/;
            if (!nicPattern.test(nic)) {
                Swal.fire({
                    icon: 'error',
                    title: 'වැරදි හැඳුනුම්පත් අංකයක්!',
                    text: 'කරුණාකර නිවැරදි NIC අංකයක් ඇතුලත් කරන්න.',
                    confirmButtonColor: '#d33'
                });
                return;
            }

            // LocalStorage තුල දත්ත තැන්පත් කිරීම
            let reqs = JSON.parse(localStorage.getItem('reqs')) || [];
            reqs.push({ name, phone, nic, amount, bank, status: 'Pending' });
            localStorage.setItem('reqs', JSON.stringify(reqs));

            // පණිවිඩය සහ ඔයාගේ WhatsApp සබැඳිය (0769233462)
            let msg = `නව ණය අයදුම්පතක්!\n\nනම: ${name}\nදුරකථන: ${phone}\nNIC: ${nic}\nමුදල: රු. ${amount}\nබැංකුව: ${bank}`;
            let whatsappUrl = `https://whatsapp.com{encodeURIComponent(msg)}`;

            // Form එක හිස් කිරීම
            document.getElementById('loanForm').reset();

            // 4. ලස්සන Popup එක පෙන්වීම
            Swal.fire({
                icon: 'success',
                title: 'සාර්ථකයි!',
                text: 'ඔබේ අයදුම්පත පටිගත කරගන්නා ලදී. දැන් ඔබව WhatsApp වෙත යොමු කරනු ඇත.',
                confirmButtonText: 'ඉදිරියට යන්න',
                confirmButtonColor: '#25D366',
                allowOutsideClick: false
            }).then((result) => {
                if (result.isConfirmed) {
                    window.open(whatsappUrl, '_blank');
                }
            });
        });
    </script>
